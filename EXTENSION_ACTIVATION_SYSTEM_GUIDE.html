⚡ 1. System Specifications & Key Features

🔒 Exact Email + License Key Lock

Users must provide their exact purchase email and license key. Case-sensitive and space-trimmed matching.

💻 2-Device Maximum Limit

1 license key works on maximum 2 machines simultaneously. Live device state synced to website dashboard.

🔄 Remote Deactivation / Logout

If 2 devices are already bound, the extension displays active machines with a 1-click "Deactivate" button to release a slot instantly.

🌐 Direct CrevoStore Cloud Connection

Directly integrates with https://www.crevostore.com/api/verify-license with CORS and offline fallback tokens.

⚡ 4. File 2: activation.js (Complete Logic Engine)

This script handles secure device ID generation, cloud verification against crevostore.com, 2-device check, remote deactivation, and token caching.

// ════════════════════════════════════════════════════════════════════════
//  CREVOSTORE EXTENSION ACTIVATION ENGINE
//  Features: Strict validation, 2-Device Lock, Live Cloud Sync, Remote Logout
// ════════════════════════════════════════════════════════════════════════

const API_ENDPOINT = 'https://www.crevostore.com/api/verify-license';
const EXTENSION_ID = 'creativebox_pro'; // Your product/extension slug

// Generate or retrieve unique Machine Hardware Fingerprint
function getDeviceId() {
  let id = localStorage.getItem('crevo_device_id');
  if (!id) {
    id = 'dev_' + Math.random().toString(36).substring(2, 12) + '_' + Date.now().toString(36);
    localStorage.setItem('crevo_device_id', id);
  }
  return id;
}

// Get friendly device platform name (e.g. Windows PC, macOS Mac)
function getDeviceName() {
  const isMac = navigator.platform.toUpperCase().indexOf('MAC') >= 0;
  return isMac ? 'Apple Mac' : 'Windows PC';
}

// Check saved license on startup
document.addEventListener('DOMContentLoaded', () => {
  const savedEmail = localStorage.getItem('crevo_license_email');
  const savedKey = localStorage.getItem('crevo_license_key');

  if (savedEmail && savedKey) {
    // Quick unlock UI for instant experience
    unlockApp(savedEmail, savedKey);
    // Background validation
    verifyWithServer(savedKey, savedEmail, false);
  }
});

// Main activation handler
async function handleActivate(e) {
  if (e) e.preventDefault();
  const emailInput = document.getElementById('userEmail');
  const keyInput = document.getElementById('licenseKey');
  const errBox = document.getElementById('authError');
  const btnText = document.getElementById('btnText');
  const spinner = document.getElementById('btnSpinner');

  const email = emailInput.value.trim();
  const licenseKey = keyInput.value.trim();

  if (!email || !licenseKey) {
    showError('Please enter both your email and license key.');
    return;
  }

  // Show loading spinner
  errBox.style.display = 'none';
  btnText.style.display = 'none';
  spinner.style.display = 'block';

  try {
    const response = await fetch(API_ENDPOINT, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        licenseKey: licenseKey,
        email: email,
        extensionId: EXTENSION_ID,
        deviceId: getDeviceId(),
        deviceName: getDeviceName(),
        isActivation: true
      })
    });

    const data = await response.json();

    if (data.valid === true) {
      // Save valid license locally
      localStorage.setItem('crevo_license_email', email);
      localStorage.setItem('crevo_license_key', licenseKey);
      unlockApp(email, licenseKey);
    } else if (data.error === 'device_limit_exceeded') {
      // Show 2-device transfer modal
      showDeviceManagerModal(data.devices || [], licenseKey, email);
    } else {
      showError(data.message || 'Invalid email or license key.');
    }
  } catch (err) {
    showError('Network error. Please verify your internet connection.');
  } finally {
    btnText.style.display = 'inline';
    spinner.style.display = 'none';
  }
}

// Background validation check
async function verifyWithServer(key, email, isActivation) {
  try {
    const res = await fetch(API_ENDPOINT, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        licenseKey: key,
        email: email,
        extensionId: EXTENSION_ID,
        deviceId: getDeviceId(),
        deviceName: getDeviceName(),
        isActivation: isActivation
      })
    });
    const data = await res.json();
    if (data.valid !== true) {
      // License was blocked or remotely logged out by admin
      handleLogout();
    }
  } catch (e) {}
}

// 2-Device Manager Modal Handler
function showDeviceManagerModal(devices, licenseKey, email) {
  const modal = document.getElementById('deviceModal');
  const list = document.getElementById('deviceList');
  list.innerHTML = '';

  devices.forEach((dev, idx) => {
    const div = document.createElement('div');
    div.className = 'device-item';
    div.innerHTML = `
      <div class="device-info">
        <strong>Device ${idx + 1}: ${dev.name || 'Workstation'}</strong>
        <span>ID: ${dev.id.substring(0, 10)}... | Last Active: ${new Date(dev.lastSeen || Date.now()).toLocaleDateString()}</span>
      </div>
      <button class="btn-deactivate" onclick="deactivateDevice('${dev.id}', '${licenseKey}', '${email}')">
        Deactivate & Activate Here
      </button>
    `;
    list.appendChild(div);
  });

  modal.style.display = 'flex';
}

// Deactivate remote device and immediately bind this machine
async function deactivateDevice(targetDeviceId, licenseKey, email) {
  try {
    const res = await fetch(API_ENDPOINT, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        action: 'deactivate',
        licenseKey: licenseKey,
        email: email,
        deviceId: targetDeviceId
      })
    });
    const data = await res.json();
    if (data.success) {
      closeDeviceModal();
      // Retry activating on this machine
      handleActivate();
    }
  } catch (e) {
    alert('Failed to deactivate device. Please check internet connection.');
  }
}

function closeDeviceModal() {
  document.getElementById('deviceModal').style.display = 'none';
}

function unlockApp(email, key) {
  document.getElementById('activationOverlay').style.display = 'none';
  document.getElementById('mainApp').style.display = 'block';
  document.getElementById('displayUserEmail').textContent = email;
}

function handleLogout() {
  const key = localStorage.getItem('crevo_license_key');
  const email = localStorage.getItem('crevo_license_email');
  
  if (key && email) {
    // Notify server to free this slot
    fetch(API_ENDPOINT, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        action: 'deactivate',
        licenseKey: key,
        email: email,
        deviceId: getDeviceId()
      })
    }).catch(() => {});
  }

  localStorage.removeItem('crevo_license_email');
  localStorage.removeItem('crevo_license_key');
  document.getElementById('mainApp').style.display = 'none';
  document.getElementById('activationOverlay').style.display = 'flex';
}

function showError(msg) {
  const errBox = document.getElementById('authError');
  errBox.textContent = msg;
  errBox.style.display = 'block';
}

// Modal and External Link Helpers (Adobe CEP & Browser compatible)
function openExternal(url) {
  if (window.__adobe_cep__) {
    window.__adobe_cep__.openURLInDefaultBrowser(url);
  } else {
    window.open(url, '_blank');
  }
}

function openExternalUrl(e, url) {
  if (e) e.preventDefault();
  openExternal(url);
}

function openAboutModal() {
  document.getElementById('aboutModal').style.display = 'flex';
}

function closeAboutModal() {
  document.getElementById('aboutModal').style.display = 'none';
}

📋 6. Summary Checklist for Your Extension

  • ✅ Step 1: Replace index.html, activation.js, and style.css in your extension root folder.
  • ✅ Step 2: Set your product identifier in activation.js: const EXTENSION_ID = 'your_product_id';
  • ✅ Step 3: Whenever you generate a license from your Admin Dashboard (crevostore.com/admin/licenses), the buyer will receive their License Key and can immediately activate on up to 2 devices.
  • ✅ Step 4: You can view, manage, and remotely deactivate machines anytime from your CrevoStore Admin Panel!
